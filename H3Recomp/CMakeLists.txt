# The runtime: kernel, GPU, audio, input and window code (H3Runtime, a
# library of our code only), linked with the recompiled game (H3RecompLib,
# the modules) into the executable H3Recomp. See docs/ARCHITECTURE.md.
#
# H3Runtime is what a release ships pre-built: on the player's machine,
# h3setup links it with the game code it compiled there (docs/ISO_TOOL.md).
# Linked whole: its hooks are strong definitions that replace generated
# weak ones, which a linker wouldn't pull out of an archive on its own.

# Decompiled game code that runs in the game (H3Native/): everything but
# the standalone cache file reader (H3NativeCache) and the generated tag
# code (H3NativeTags), which are libraries of their own.
file(GLOB_RECURSE H3_NATIVE_SOURCES CONFIGURE_DEPENDS ${PROJECT_SOURCE_DIR}/H3Native/*.cpp)
list(FILTER H3_NATIVE_SOURCES EXCLUDE REGEX "/H3Native/(cache/cache_file\\.cpp|tags/generated/)")

add_executable(H3Recomp
    # Weak fallbacks for every import, generated by h3xexinfo. Linked as an
    # object (not via a static library) so strong definitions always win.
    # The DLLs' stubs come in through H3_MODULE_STUBS the same way.
    ${H3_PPC_DIR}/import_stubs.cpp
    # Table of recompiled DLLs, generated by H3RecompLib/CMakeLists.txt.
    ${H3_MODULE_REGISTRY}
)
target_include_directories(H3Recomp PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(H3Recomp PRIVATE $<LINK_LIBRARY:WHOLE_ARCHIVE,H3Runtime> H3RecompLib ${H3_MODULES} ${H3_MODULE_STUBS})

target_include_directories(H3Runtime PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
# xxHash (header only, vendored by XenonRecomp): fast hashing of texture memory.
target_include_directories(H3Runtime PRIVATE ${PROJECT_SOURCE_DIR}/thirdparty/XenonRecomp/thirdparty/xxHash)
# XXH3's AVX2 build, picked at runtime (ContentHash) so the executable
# still runs on CPUs without AVX2.
if (WIN32)
    target_sources(H3Runtime PRIVATE platform/windows_compat.cpp)
endif()

if (MSVC)
    set_source_files_properties(gpu/renderer/content_hash_avx2.cpp PROPERTIES COMPILE_OPTIONS /arch:AVX2)
else()
    set_source_files_properties(gpu/renderer/content_hash_avx2.cpp PROPERTIES COMPILE_OPTIONS -mavx2)
endif()

find_package(SDL3 REQUIRED CONFIG)
# shaderc (runtime GLSL -> SPIR-V): pkg-config on Linux; on Windows the
# static shaderc_combined from tools/build_windows_deps.sh (H3_WINDOWS_DEPS).
if (WIN32)
    find_library(H3_SHADERC_LIBRARY shaderc_combined PATHS "${H3_WINDOWS_DEPS}/lib" NO_DEFAULT_PATH REQUIRED)
    add_library(H3Shaderc INTERFACE)
    target_include_directories(H3Shaderc INTERFACE "${H3_WINDOWS_DEPS}/include")
    target_link_libraries(H3Shaderc INTERFACE "${H3_SHADERC_LIBRARY}")
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SHADERC REQUIRED IMPORTED_TARGET shaderc)
    add_library(H3Shaderc INTERFACE)
    target_link_libraries(H3Shaderc INTERFACE PkgConfig::SHADERC)
endif()

# Counts and samples calls into the statically linked D3D library (to name
# its functions). Writes d3d_trace.txt at frame 150.
option(H3_TRACE_D3D "Trace calls into the game's D3D library (debugging only)" OFF)
if (H3_TRACE_D3D)
    target_sources(H3Runtime PRIVATE debug/d3d_trace.cpp debug/d3d_trace_hooks.cpp)
endif()

# H3RecompLib for its headers (ppc_recomp_shared.h: the generated
# functions the hooks replace and call); the executable links its code.
target_link_libraries(H3Runtime PUBLIC H3Shader H3Config H3NativeCache H3NativeTags H3Shaderc SDL3::SDL3 PRIVATE H3RecompLib)

# FFmpeg with the xmaframes decoder (XMA audio), built by tools/build_ffmpeg.sh
# (on Windows, tools/build_windows_deps.sh).
if (WIN32)
    set(H3_FFMPEG_DIR "${H3_WINDOWS_DEPS}/ffmpeg")
    set(H3_FFMPEG_SCRIPT tools/build_windows_deps.sh)
else()
    set(H3_FFMPEG_DIR "${PROJECT_SOURCE_DIR}/build/ffmpeg")
    set(H3_FFMPEG_SCRIPT tools/build_ffmpeg.sh)
endif()
if (NOT EXISTS "${H3_FFMPEG_DIR}/lib/libavcodec.a")
    message(FATAL_ERROR "XMA audio needs FFmpeg with the xmaframes decoder: run ${H3_FFMPEG_SCRIPT} first")
endif()
target_include_directories(H3Runtime BEFORE PRIVATE "${H3_FFMPEG_DIR}/include")
target_link_libraries(H3Runtime PUBLIC "${H3_FFMPEG_DIR}/lib/libavcodec.a" "${H3_FFMPEG_DIR}/lib/libavutil.a")
if (WIN32)
    target_link_libraries(H3Runtime PUBLIC bcrypt ws2_32)  # FFmpeg's random seeds; system link's socket
else()
    target_link_libraries(H3Runtime PUBLIC m)
endif()

# Windows: the DLLs it loads (SDL3, the C++ runtime) go next to the executable.
if (WIN32)
    add_custom_command(TARGET H3Recomp POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_RUNTIME_DLLS:H3Recomp> ${H3_VCREDIST_DLLS} $<TARGET_FILE_DIR:H3Recomp>
        COMMAND_EXPAND_LISTS)
endif()

# The release version and the commit this is built from: system link plays
# with games of the same version (and game executable and 60 Hz fixes;
# net/link.cpp); the commit is logged. A build with local changes is marked
# "-dirty".
execute_process(COMMAND git describe --always --dirty --abbrev=12 WORKING_DIRECTORY ${PROJECT_SOURCE_DIR}
    OUTPUT_VARIABLE H3_GIT_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if (NOT H3_GIT_COMMIT)
    set(H3_GIT_COMMIT unknown)
endif()
set_source_files_properties(net/link.cpp PROPERTIES COMPILE_DEFINITIONS
    "H3_GIT_COMMIT=\"${H3_GIT_COMMIT}\";H3_VERSION=\"${PROJECT_VERSION}\"")
