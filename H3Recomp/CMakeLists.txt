# The runtime executable: kernel, GPU, audio, input and window code linked
# with the recompiled game (H3RecompLib). See docs/ARCHITECTURE.md.

# Decompiled game code that runs in the game (H3Native/): everything but
# the standalone cache file reader (H3NativeCache) and the generated tag
# code (H3NativeTags), which are libraries of their own.
file(GLOB_RECURSE H3_NATIVE_SOURCES CONFIGURE_DEPENDS ${PROJECT_SOURCE_DIR}/H3Native/*.cpp)
list(FILTER H3_NATIVE_SOURCES EXCLUDE REGEX "/H3Native/(cache/cache_file\\.cpp|tags/generated/)")

add_executable(H3Recomp
    main.cpp
    memory.cpp
    gpu/command_processor.cpp
    gpu/frame_capture.cpp
    gpu/renderer.cpp
    gpu/renderer/content_hash_avx2.cpp
    gpu/renderer/copy.cpp
    gpu/renderer/debug.cpp
    gpu/renderer/draw.cpp
    gpu/renderer/memexport.cpp
    gpu/renderer/queries.cpp
    gpu/renderer/pipelines.cpp
    gpu/renderer/postprocess.cpp
    gpu/renderer/resolve.cpp
    gpu/renderer/shaders.cpp
    gpu/renderer/targets.cpp
    gpu/renderer/textures.cpp
    gpu/texture.cpp
    input/input.cpp
    debug/first_calls.cpp
    debug/profiler.cpp
    debug/state_hash.cpp
    debug/write_watch.cpp
    hooks/block_copy.cpp
    hooks/campaign.cpp
    hooks/fibers.cpp
    hooks/random_seed.cpp
    hooks/tick_rate.cpp
    hooks/xma.cpp
    kernel/imports.cpp
    kernel/module.cpp
    kernel/modules.cpp
    kernel/objects.cpp
    kernel/thread.cpp
    kernel/variables.cpp
    kernel/vfs.cpp
    kernel/xam.cpp
    kernel/xam_user.cpp
    kernel/xam_content.cpp
    kernel/xam_notify.cpp
    kernel/xnet.cpp
    apu/audio_system.cpp
    apu/xma_decoder.cpp
    kernel/xboxkrnl.cpp
    kernel/xboxkrnl_audio.cpp
    kernel/xboxkrnl_crypt.cpp
    kernel/xboxkrnl_io.cpp
    kernel/xboxkrnl_memory.cpp
    kernel/xboxkrnl_modules.cpp
    kernel/xboxkrnl_objects.cpp
    kernel/xboxkrnl_rtl.cpp
    kernel/xboxkrnl_strings.cpp
    kernel/xboxkrnl_threading.cpp
    kernel/xboxkrnl_video.cpp
    video/bink_view.cpp
    video/window.cpp
    # Decompiled game code (H3Native/, globbed below), replacing recompiled functions.
    ${H3_NATIVE_SOURCES}
    # Weak fallbacks for every import, generated by h3xexinfo. Linked as an
    # object (not via a static library) so strong definitions always win.
    # The DLLs' stubs come in through H3_MODULE_STUBS the same way.
    ${H3_PPC_DIR}/import_stubs.cpp
    # Table of recompiled DLLs, generated by H3RecompLib/CMakeLists.txt.
    ${H3_MODULE_REGISTRY}
)

target_include_directories(H3Recomp PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
# xxHash (header only, vendored by XenonRecomp): fast hashing of texture memory.
target_include_directories(H3Recomp PRIVATE ${PROJECT_SOURCE_DIR}/thirdparty/XenonRecomp/thirdparty/xxHash)
# XXH3's AVX2 build, picked at runtime (ContentHash) so the executable
# still runs on CPUs without AVX2.
set_source_files_properties(gpu/renderer/content_hash_avx2.cpp PROPERTIES COMPILE_OPTIONS -mavx2)

find_package(SDL3 REQUIRED CONFIG)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SHADERC REQUIRED IMPORTED_TARGET shaderc)

# Counts and samples calls into the statically linked D3D library (to name
# its functions). Writes d3d_trace.txt at frame 150.
option(H3_TRACE_D3D "Trace calls into the game's D3D library (debugging only)" OFF)
if (H3_TRACE_D3D)
    target_sources(H3Recomp PRIVATE debug/d3d_trace.cpp debug/d3d_trace_hooks.cpp)
endif()

target_link_libraries(H3Recomp PRIVATE H3RecompLib H3Shader H3Config H3NativeCache H3NativeTags PkgConfig::SHADERC SDL3::SDL3 ${H3_MODULES} ${H3_MODULE_STUBS})

# FFmpeg with the xmaframes decoder (XMA audio), built by tools/build_ffmpeg.sh.
set(H3_FFMPEG_DIR "${PROJECT_SOURCE_DIR}/build/ffmpeg")
if (NOT EXISTS "${H3_FFMPEG_DIR}/lib/libavcodec.a")
    message(FATAL_ERROR "XMA audio needs FFmpeg with the xmaframes decoder: run tools/build_ffmpeg.sh first")
endif()
target_include_directories(H3Recomp BEFORE PRIVATE "${H3_FFMPEG_DIR}/include")
target_link_libraries(H3Recomp PRIVATE "${H3_FFMPEG_DIR}/lib/libavcodec.a" "${H3_FFMPEG_DIR}/lib/libavutil.a" m)
